# 初三全科 AI 学习助手（网页版 MVP）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建一个网页版 MVP——初三学生可自主使用的全科 AI 学习助手，含 AI 答疑讲解、智能刷题、错题本、游戏化系统与鼓励型 AI 陪伴角色，用 DeepSeek API 驱动。

**Architecture:** 轻量前端网页（React + Vite）通过后端服务（Node/Express）调用 DeepSeek API，后端负责 AI 编排、用户数据（SQLite）、积分/错题/打卡逻辑。前端与后端通过 REST JSON 接口通信。数据持久化用 SQLite（prisma 或原生 sqlite3），不过度设计。

**Tech Stack:** React 18 + Vite + Tailwind CSS；Node.js + Express；DeepSeek Chat API（OpenAI 兼容协议）；SQLite（better-sqlite3）；JWT 简易登录。

## Global Constraints

- 产品气质四位一体：游戏化 + AI 陪伴 + 即时正反馈 + 轻量不压迫（源自 spec 第 1 节）。
- AI 永远先鼓励再指正，语气像朋友（源自 spec 第 5 节）。
- 不布置作业、不弹窗催促；排行榜默认可关闭（源自 spec 第 5 节）。
- MVP 仅网页版，不做小程序/App（源自 spec 第 4/6 节）。
- AI 能力全部来自 DeepSeek API，不自研模型（源自 spec 第 4 节）。
- 数据存储先用 SQLite，不过度设计（源自 spec 第 4 节）。

- [ ] **Step 4: 实现 auth.js（明文密码仅 MVP，注释标注生产需哈希）**
```js
import jwt from 'jsonwebtoken';
import { db } from './db.js';
const SECRET = process.env.JWT_SECRET || 'dev-secret';
// MVP 用明文；生产必须 bcrypt 哈希
export function register(username, password) {
  const ins = db.prepare('INSERT INTO users (username,password) VALUES (?,?)');
  const info = ins.run(username, password);
  return { token: jwt.sign({ id: info.lastInsertRowid }, SECRET), userId: info.lastInsertRowid };
}
export function login(username, password) {
  const row = db.prepare('SELECT id,password FROM users WHERE username=?').get(username);
  if (!row || row.password !== password) throw new Error('invalid credentials');
  return { token: jwt.sign({ id: row.id }, SECRET), userId: row.id };
}
export function authMiddleware(req, res, next) {
  const h = req.headers.authorization || '';
  try { req.userId = jwt.verify(h.replace('Bearer ', ''), SECRET).id; next(); }
  catch { res.status(401).json({ error: 'unauthorized' }); }
}
```

```js
import OpenAI from 'openai';
const client = new OpenAI({
  apiKey: process.env.DEEPSEEK_API_KEY,
  baseURL: process.env.DEEPSEEK_BASE_URL || 'https://api.deepseek.com'
});
const SYSTEM = `你是"阿杰学长"，一个初三学生的 AI 学霸同桌。语气亲切像朋友，永远先鼓励再指正。
讲解题目用引导式：先问思路，再点破关键，最后总结方法。绝不直接把答案硬灌。
每次回复 JSON：{"reply":"讲解内容","encouragement":"一句鼓励"}。`;

export async function explainQuestion({ subject, question, history = [] }) {
  const msgs = [{ role: 'system', content: SYSTEM }, ...history, { role: 'user', content: `科目:${subject}\n题目:${question}` }];
  const r = await client.chat.completions.create({ model: 'deepseek-chat', messages: msgs, response_format: { type: 'json_object' } });
  return JSON.parse(r.choices[0].message.content);
}
export async function generateQuiz({ subject, knowledgePoint, difficulty = 2 }) {
  const msgs = [{ role: 'system', content: SYSTEM + ' 出题请返回 JSON:{"question","options":["A..","B.."],"answer":"A","explanation","difficulty"}' },
    { role: 'user', content: `科目:${subject} 考点:${knowledgePoint} 难度:${difficulty}/5，出一道选择题` }];
  const r = await client.chat.completions.create({ model: 'deepseek-chat', messages: msgs, response_format: { type: 'json_object' } });
  return JSON.parse(r.choices[0].message.content);
}
export async function analyzeWeakness(records) {
  const msgs = [{ role: 'system', content: SYSTEM + ' 根据做题记录返回 JSON:{"redLight":["薄弱考点"],"greenLight":["掌握考点"]}' },
    { role: 'user', content: '记录:' + JSON.stringify(records) }];
  const r = await client.chat.completions.create({ model: 'deepseek-chat', messages: msgs, response_format: { type: 'json_object' } });
  return JSON.parse(r.choices[0].message.content);
}
```

import { db } from './db.js';
export function addPoints(userId, delta) {
  const u = db.prepare('UPDATE users SET points = points + ? WHERE id=? RETURNING points').get(delta, userId);
  const level = Math.floor(u.points / 100) + 1;
  db.prepare('UPDATE users SET level=? WHERE id=?').run(level, userId);
  return { points: u.points, level };
}
export function checkIn(userId) {
  const today = new Date().toISOString().slice(0,10);
  const exists = db.prepare('SELECT 1 FROM checkins WHERE user_id=? AND day=?').get(userId, today);
  if (exists) return { streak: db.prepare('SELECT streak FROM users WHERE id=?').get(userId).streak, rewarded: false };
  db.prepare('INSERT INTO checkins (user_id,day) VALUES (?,?)').run(userId, today);
  const before = db.prepare('SELECT streak FROM users WHERE id=?').get(userId).streak;
  const streak = before + 1;
  db.prepare('UPDATE users SET streak=? WHERE id=?').run(streak, userId);
  return { streak, rewarded: true };
}
export function earnBadge(userId, badge) {
  // MVP: 简单记录到 username 外的 badge 列可后续扩展；此处返回 true 占位
  return true;
}

// src/server/index.js
import express from 'express';
import { router as authRouter, authMiddleware } from './auth.js';
import { explainQuestion } from './ai.js';
import { generateQuiz } from './ai.js';
import { addPoints, checkIn } from './gamify.js';
import { addMistake, getMistakes, getWeakness } from './mistakes.js';
import { db } from './db.js';

const app = express();
app.use(express.json());
app.use('/api', authRouter);

app.post('/api/chat', authMiddleware, async (req,res)=>{
  try {
    const r = await explainQuestion(req.body);
    res.json(r);
  } catch(e){ res.status(500).json({error:e.message}); }
});

app.post('/api/quiz/next', authMiddleware, async (req,res)=>{
  try { res.json(await generateQuiz(req.body)); } catch(e){ res.status(500).json({error:e.message}); }
});

app.post('/api/quiz/answer', authMiddleware, async (req,res)=>{
  const { userId } = req;
  const { subject, knowledgePoint, correct, difficulty } = req.body;
  db.prepare('INSERT INTO quiz_records (user_id,subject,knowledge_point,correct,difficulty) VALUES (?,?,?,?,?)')
    .run(userId, subject, knowledgePoint, correct?1:0, difficulty||2);
  if (correct) { const g = addPoints(userId, 10); if(!correct) addMistake(userId,{subject,knowledgePoint,question:req.body.question,answer:req.body.answer}); res.json({...g}); }
  else { addMistake(userId,{subject,knowledgePoint,question:req.body.question,answer:req.body.answer}); res.json({points:0,level:1}); }
});

app.get('/api/progress', authMiddleware, async (req,res)=>{
  const u = db.prepare('SELECT points,level,streak FROM users WHERE id=?').get(req.userId);
  const wk = await getWeakness(req.userId);
  res.json({ ...u, weakness: wk });
});

app.get('/api/mistakes', authMiddleware, (req,res)=> res.json(getMistakes(req.userId)));

app.listen(process.env.PORT || 3001, ()=> console.log('server up'));
